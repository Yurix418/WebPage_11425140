import hmac, hashlib, time, requests

API_KEY = "makebit_live_api_key_888"
SECRET = b"makebit_secret_token_11425140"

def place_order(symbol, side, price, qty):
    timestamp = str(int(time.time() * 1000))
    query = f"symbol={symbol}&side={side}&type=LIMIT&price={price}&qty={qty}&ts={timestamp}"
    signature = hmac.new(SECRET, query.encode(), hashlib.sha256).hexdigest()
    
    headers = {"X-MB-KEY": API_KEY, "X-MB-SIGN": signature}
    url = f"https://api.makebit.com/v1/order?{query}"
    return requests.post(url, headers=headers).json()
